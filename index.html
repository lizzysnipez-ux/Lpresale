async function buyTokens() { if (redirectToPhantomIfMobile()) return; if (!publicKey) return; const amount = parseFloat(document.getElementById('solAmountInput').value); if (amount < 0.5 || amount > 500) return; const btn = document.getElementById('buyTokenBtn'); const msg = document.getElementById('transactionMessage'); btn.disabled = true; btn.innerText = 'Processing...'; msg.classList.add('hidden'); const MAX_RETRIES = 2; let lastError = null; for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) { try { // 1. ALWAYS fetch a FRESH blockhash for each attempt const { blockhash, lastValidBlockHeight, context } = await tryRpc(conn => conn.getLatestBlockhashAndContext('confirmed') ); if (!blockhash) throw new Error('Failed to fetch blockhash'); // 2. Build a brand-new transaction with this fresh blockhash const tx = new solanaWeb3.Transaction({ recentBlockhash: blockhash, feePayer: new solanaWeb3.PublicKey(publicKey) }); tx.add(solanaWeb3.ComputeBudgetProgram.setComputeUnitPrice({ microLamports: PRIORITY_FEE })); tx.add(solanaWeb3.SystemProgram.transfer({ fromPubkey: new solanaWeb3.PublicKey(publicKey), toPubkey: new solanaWeb3.PublicKey(DEV_WALLET), lamports: Math.floor(amount * solanaWeb3.LAMPORTS_PER_SOL) })); // 3. Sign and send. The minContextSlot is crucial for Phantom on Android. const { signature } = await wallet.signAndSendTransaction(tx, { minContextSlot: context.slot }); // 4. Confirm using the SAME blockhash and lastValidBlockHeight const conn = connection || new solanaWeb3.Connection(RPC_LIST[0], 'confirmed'); await conn.confirmTransaction( { signature, blockhash, lastValidBlockHeight }, 'confirmed' ); // --- SUCCESS: Update UI and exit the function --- totalRaised += amount; buyerCount++; updateStatsUI(); addRecentPurchase(amount, publicKey.slice(-4)); msg.classList.remove('hidden'); msg.innerHTML = `
✓ Success! ${Math.floor(amount * getCurrentTier().tokensPerSol).toLocaleString()} LIZZY reserved. View tx
`; setTimeout(() => { msg.innerHTML = ''; msg.classList.add('hidden'); }, 6000); await fetchBalance(); return; // Exit on success } catch (err) { lastError = err; console.error(`❌ Attempt ${attempt + 1} failed:`, err); // If it's a stale blockhash error, loop and retry with a fresh one if (err.message?.includes('blockhash') || err.message?.includes('expired')) { console.warn('Blockhash stale, retrying with a fresh one...'); continue; // Go to the next iteration of the loop } else { // For any other error (e.g., user rejected), break immediately break; } } } // --- ALL ATTEMPTS FAILED: Show the final error --- msg.classList.remove('hidden'); let errorMsg = lastError?.message || 'Transaction failed after multiple attempts.'; if (lastError?.code === 4001) errorMsg = 'Transaction rejected in wallet.'; msg.innerHTML = `
✗ ${errorMsg.slice(0, 100)}
`; setTimeout(() => { msg.innerHTML = ''; msg.classList.add('hidden'); }, 6000); // Re-enable the button btn.disabled = false; btn.innerText = 'Buy $LIZZY'; updateTokenUI(); }
